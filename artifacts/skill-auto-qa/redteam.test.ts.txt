  })
  test("valid inputs accepted and frozen; absent defaults to shadow; null default", () => {
    const p = parseSkillAuto({ mode: "auto", include: ["a_b-1"], exclude: ["x"] }) as any
    expect(p.mode).toBe("auto"); expect(p.include.has("a_b-1")).toBe(true); expect(Object.isFrozen(p)).toBe(true)
    expect((parseSkillAuto(undefined) as any).mode).toBe("shadow")
    expect((parseSkillAuto({}) as any).mode).toBe("shadow")
    expect((parseSkillAuto({ include: Array(256).fill("a") }) as any).mode).toBe("shadow")
  })
  test("readSkillTrust: invalid/hostile files -> off", () => {
    const file = join(root, "otter.jsonc")
    const read = (t: string) => readSkillTrust(file, root)
    for (const t of ['{"skill_auto":{"mode":"auto","bad":1}}', '{"skill_auto":"auto"}', '{"skill_auto":{"mode":"auto","include":["../x"]}}', "{ broken"]) {
      writeFileSync(file, t); const r = read(t)
      expect(r.auto.mode, t).toBe("off"); expect(r.problem, t).toBeTruthy()
    }
    writeFileSync(file, '{"skill_auto":{"mode":"auto"}, "skill_trust":"x"}'); expect(read("").problem).toBeTruthy() // independent fields: bad trust trusts nothing, auto unaffected
    for (const t of ["[]", "null", '"s"']) { writeFileSync(file, t); expect(read(t).auto.mode).toBe("shadow") } // non-object file == absent => non-applying default
    writeFileSync(file, '{"skill_auto":{"mode":"auto","include":["a"]}}'); expect(read("").auto.mode).toBe("auto")
    writeFileSync(file, '{"skill_auto":{"mode":"auto"},"skill_auto":{"mode":"off"}}'); expect(typeof read("").auto.mode).toBe("string")
    expect(readSkillTrust(join(root, "missing.jsonc"), root).auto.mode).toBe("shadow")
    expect(readSkillTrust(root, root).auto.mode).toBe("off") // directory -> unreadable
  })
})

describe("4 prefilterSkills", () => {
  const many = (n: number, desc = (i: number) => `identical description ${i % 2}`) => Array.from({ length: n }, (_, i) => ({ name: `s${String(i).padStart(2, "0")}`, description: desc(i) }))
  test("determinism under permutation, limit, ties", () => {
    const skills = many(30, () => "same words here")
    const a = prefilterSkills("same words", skills).map(s => s.name)
    for (let k = 0; k < 20; k++) {
      const shuffled = [...skills].sort(() => Math.random() - 0.5)
      expect(prefilterSkills("same words", shuffled).map(s => s.name)).toEqual(a)
    }
    expect(a).toHaveLength(PREFILTER_LIMIT); expect(a).toEqual([...a].sort())
  })
  test("limit variants", () => {
    const skills = many(12)
    expect(prefilterSkills("x", skills, 0)).toHaveLength(0)
    expect(prefilterSkills("x", skills, 1)).toHaveLength(1)
    expect(prefilterSkills("x", skills, 100)).toHaveLength(12)
    expect(prefilterSkills("x", [], 8)).toEqual([])
  })
  test("empty / emoji / mixed / punctuation requests", () => {
    const skills = many(20, i => `위키 정리 llm wiki ${i}`)
    for (const q of ["", "   ", "😀😀😀", "!!!???", "한", "a", "wiki😀위키", "ＬＬＭ", "\u0000\u200b", "x".repeat(2000), "위".repeat(2000), "ǅ İ ß"]) {
      const r = prefilterSkills(q, skills); expect(r).toHaveLength(PREFILTER_LIMIT)
      expect(prefilterSkills(q, [...skills].reverse()).map(s => s.name)).toEqual(r.map(s => s.name))
    }
    expect(prefilterSkills("", skills).map(s => s.name)).toEqual(skills.map(s => s.name).sort().slice(0, 8))
  })
  test("returns same object identities, does not mutate input, ranks relevant first", () => {
    const skills = many(20); skills[13] = { name: "zeta-wiki", description: "compile llm wiki pages 위키" }
    const copy = [...skills]
    const r = prefilterSkills("위키 compile", skills)
    expect(skills).toEqual(copy); expect(r[0]).toBe(skills[13]); expect(r).toHaveLength(8)
  })
  test("duplicate names do not crash; NaN-safe with huge descriptions", () => {
    const skills = [...many(10), ...many(10)]; expect(prefilterSkills("s01 identical", skills)).toHaveLength(8)
    const big = many(10, () => "word ".repeat(20000)); expect(prefilterSkills("word", big)).toHaveLength(8)
  })
  test("duplicate/prototype-ish names", () => {
    const skills = many(9); skills[0] = { name: "__proto__", description: "constructor toString" }
    expect(prefilterSkills("constructor toString", skills)[0]!.name).toBe("__proto__")
  })
})

describe("5 backoff", () => {
  test("only transport errors count; others reset; caller abort not counted; cooldown resets", async () => {
    let clock = 1000
    const s = createAutoSkillSelector(deps, { now: () => clock })
    const down = () => new Response("down", { status: 503 })
    // non-transport failures never trigger backoff
    for (const h of [() => Response.json(ans("none", { none: 0.99, plan: 0.01 })), () => Response.json(ans("plan", { plan: 0.6, none: 0.4 })), () => Response.json({ answers: {} })]) {
      handler = h
      for (let i = 0; i < AUTO_SELECT_FAILURE_LIMIT + 2; i++) { const r = await sel(s); expect(r.reason).not.toBe("backoff") }
    }
    // interleave: 2 failures, success, 2 failures -> no backoff
    handler = down; await sel(s); await sel(s)
    handler = () => Response.json(ans("plan", { plan: 0.99, none: 0.01 })); expect((await sel(s)).skill).toBe("plan")
    handler = down; await sel(s); await sel(s); expect((await sel(s)).reason).toBe("unavailable")
    expect((await sel(s)).reason).toBe("backoff")
    clock += AUTO_SELECT_BACKOFF_MS - 1; expect((await sel(s)).reason).toBe("backoff")
    clock += 1; handler = down; expect((await sel(s)).reason).toBe("unavailable") // probe
    // failure counter reset after entering backoff: needs 3 more, not 1
    expect((await sel(s)).reason).toBe("unavailable"); expect((await sel(s)).reason).toBe("unavailable")
    expect((await sel(s)).reason).toBe("backoff")
  })
  test("timeouts count; caller abort mid-flight does not", async () => {
    let clock = 0; const s = createAutoSkillSelector(deps, { now: () => clock })
    deps.config.skill_router.timeout_ms = 60
    handler = () => new Promise(r => setTimeout(() => r(new Response("x")), 400))
    for (let i = 0; i < 5; i++) { const c = new AbortController(); const p = sel(s, "plan this", cands(["plan"]), c.signal); setTimeout(() => c.abort(), 10); await p }
    expect((await sel(s)).reason).toBe("aborted") // timeout #1 (not backoff)
    await sel(s); await sel(s)
    expect((await sel(s)).reason).toBe("backoff")
  })
  test("backoff and input bounds send no request; cancel() works; slot frees after timeout", async () => {
    const s = createAutoSkillSelector(deps, { now: () => 0 })
    handler = () => new Response("x", { status: 500 }); for (let i = 0; i < 3; i++) await sel(s)
    const n = requests.length; await sel(s); expect(requests.length).toBe(n)
  })
  test("concurrent call gets busy, no second request", async () => {
    const s = createAutoSkillSelector(deps)
    handler = () => new Promise(r => setTimeout(() => r(Response.json(ans("none", { none: 0.9, plan: 0.1 }))), 150))
    const p1 = sel(s); await Bun.sleep(30); const r2 = await sel(s); await p1
    expect(r2.reason).toBe("busy"); expect(requests).toHaveLength(1)
  })
})

describe("8 session B1/B2", () => {
  const signal = () => new AbortController().signal
  let fx: ReturnType<typeof tempWorktree>, home: string, store: any, sessions: SkillSession[], choices: string[]
  const git = (...a: string[]) => execFileSync("git", ["-c", "core.hooksPath=/dev/null", "-c", "commit.gpgsign=false", ...a], { cwd: fx.worktree, stdio: "pipe", env: { ...process.env, GIT_AUTHOR_NAME: "t", GIT_AUTHOR_EMAIL: "t@e.t", GIT_COMMITTER_NAME: "t", GIT_COMMITTER_EMAIL: "t@e.t", GIT_CONFIG_NOSYSTEM: "1", GIT_CONFIG_GLOBAL: "/dev/null" } })
  const put = (base: string, name: string, body: string, tools?: string) => {
    const d = join(base, ".agents", "skills", name); mkdirSync(d, { recursive: true })
    writeFileSync(join(d, "SKILL.md"), `---\nname: ${name}\ndescription: Inspect thing\n${tools ? `allowed-tools: ${tools}\n` : ""}---\n${body}\n`)
  }
  beforeEach(() => {
    fx = tempWorktree(); fx.worktree = realpathSync.native(fx.worktree); git("init", "--quiet")
    home = realpathSync.native(mkdtempSync(join(tmpdir(), "rt-home-")))
    const runtime = createCliRuntime({ worktree: fx.worktree, engine: "local", model: "fixture/model" })
    store = attachAdmission(runtime, "rt_sess", { engine: "local", model: "fixture/model" })
    sessions = []; choices = []
  })
  afterEach(() => { for (const s of sessions) s.close(); fx.cleanup(); rmSync(home, { recursive: true, force: true }) })
  const mk = async (trust: SkillTrust) => {
    const roots = await resolveSkillRoots({ projectRoot: fx.worktree, homeDirectory: home })
    const s = await SkillSession.create({ roots, store, trust, terminal: { stdinIsTTY: true, stderrIsTTY: true, choose: async (t: string) => { choices.push(t); return 0 } } }, signal())
    sessions.push(s); return s
  }
  test("B1 untracked project skill shadowing name-trusted home skill: not candidate; prepare never refuses with zero prompts; also committed-then-modified, and symlinked name", async () => {
    put(home, "orca-cli", "HOME")
    const s = await mk(parseSkillTrust(["orca-cli"], home) as SkillTrust)
    expect((await s.autoCandidates(DEFAULT_SKILL_AUTO, 1000, [], signal())).map(c => c.name)).toEqual(["orca-cli"])
    put(fx.worktree, "orca-cli", "REPO EVIL")
    expect(await s.autoCandidates(DEFAULT_SKILL_AUTO, 1000, [], signal())).toEqual([])
    await expect(s.prepare("orca-cli", signal(), { consent: "never", requiredTools: [] })).rejects.toThrow("not usable without consent")
    await expect(s.prepare("orca-cli", signal(), { consent: "never", requiredTools: ["read"] })).rejects.toThrow()
    expect(choices).toEqual([]); expect(s.listGrants()).toEqual([])
    // import include cannot rescue a project skill either
    expect(await s.autoCandidates({ ...DEFAULT_SKILL_AUTO, include: new Set(["orca-cli"]) }, 1000, [], signal())).toEqual([])
    // committed project skill with name trust only: baseline-common is allowed, but then modified working bytes are not
